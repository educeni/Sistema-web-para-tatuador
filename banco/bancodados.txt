CREATE DATABASE tatuador;
USE tatuador;

CREATE TABLE clientes(

id_cliente int NOT NULL AUTO_INCREMENT,
nome_cliente VARCHAR(25),
tel_cliente INT (14),
email_cliente VARCHAR(50),
insta_cliente VARCHAR(15),

PRIMARY KEY (id_cliente)
);

CREATE TABLE IF NOT EXISTS tatuador.materiais (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    categoria VARCHAR(50) NOT NULL,
    quantidade_atual INT NOT NULL DEFAULT 0
);

CREATE TABLE infor_cliente(

id_infor int NOT NULL AUTO_INCREMENT,
id_cliente INT,
alergias VARCHAR(25),
tipo_pele VARCHAR (25),
medicamentos VARCHAR(50),
cond_saude VARCHAR(15),

PRIMARY KEY (id_infor),
FOREIGN KEY (id_cliente) REFERENCES clientes(id_cliente)
);




CREATE TABLE agendamento(

id_agend int NOT NULL AUTO_INCREMENT,
id_cliente INT,
data_agen date,
hora_agen time,
valor_agen float,
status_agen ENUM("em andamento", "cancelado", "concluido"),

PRIMARY KEY (id_agend),
FOREIGN KEY (id_cliente) REFERENCES clientes(id_cliente)
);



CREATE TABLE despesas(

id_produto int NOT NULL AUTO_INCREMENT,
nome_prod VARCHAR(25),
valor_prod float,

PRIMARY KEY (id_produto)
);



CREATE TABLE depesas_agendamento(

id_agend INT,
id_produto INT,


FOREIGN KEY (id_produto) REFERENCES despesas(id_produto),
FOREIGN KEY (id_agend) REFERENCES agendamento(id_agend)
);
